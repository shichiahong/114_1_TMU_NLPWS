# === 匯入必要的函式庫 ===
import argparse              # 用於解析命令列參數（例如 --train, --dev, --out）
import pandas as pd          # 處理 CSV 資料的常用資料分析工具
import numpy as np           # 用於數值運算
from sklearn.model_selection import KFold          # 用於建立 K 折交叉驗證
from sklearn.feature_extraction.text import TfidfVectorizer  # 將文字轉換為 TF-IDF 向量
from sklearn.linear_model import Ridge             # Ridge Regression 模型（線性迴歸 + L2 正則化）
from sklearn.pipeline import make_pipeline         # 將多個步驟（TF-IDF + 模型）組合成 pipeline
from sklearn.metrics import mean_squared_error     # 用於計算 MSE（均方誤差）
from scipy.stats import pearsonr                   # 用於計算 Pearson 相關係數
import random                                     # 控制隨機性（與 numpy 一起設定種子）

# === 固定隨機種子以確保結果可重現 ===
def set_seed(seed=42):
    np.random.seed(seed)
    random.seed(seed)

# === 自訂 Pearson 函數：計算真實值與預測值的皮爾森相關係數 ===
def pearson(y_true, y_pred):
    # 若真實值或預測值只有一個唯一值，皮爾森無法計算，回傳 0
    if len(np.unique(y_true)) < 2 or len(np.unique(y_pred)) < 2:
        return 0.0
    # 否則計算皮爾森相關係數，取 .statistic 屬性
    return pearsonr(y_true, y_pred).statistic

# === 主訓練與評估函數 ===
def train_eval(train_path, dev_path=None, out_pred_path=None):
    set_seed(42)  # 固定隨機性
    df_tr = pd.read_csv(train_path)  # 讀取訓練資料
    text = df_tr["text"].astype(str).values     # 取出文字欄位
    y_v = df_tr["valence"].values               # 取出情感強度（Valence）
    y_a = df_tr["arousal"].values               # 取出喚起強度（Arousal）

    # === 建立兩個模型（Valence 與 Arousal 各自一個）===
    # TF-IDF 轉換器：使用 1~2 gram，忽略出現次數少於 2 次的詞，最大特徵數 200000
    # Ridge Regression：線性模型，α=2.0 為正則化強度
    model_v = make_pipeline(
        TfidfVectorizer(ngram_range=(1,2), min_df=2, max_features=200000),
        Ridge(alpha=2.0, random_state=42)
    )
    model_a = make_pipeline(
        TfidfVectorizer(ngram_range=(1,2), min_df=2, max_features=200000),
        Ridge(alpha=2.0, random_state=42)
    )

    # === 進行 5 折交叉驗證 ===
    kf = KFold(n_splits=5, shuffle=True, random_state=42)
    preds_v = np.zeros(len(df_tr))  # 儲存 Valence 的預測結果
    preds_a = np.zeros(len(df_tr))  # 儲存 Arousal 的預測結果
    for tr_idx, va_idx in kf.split(text):  # 每一折的訓練與驗證索引
        # 訓練 Valence 模型
        model_v.fit(text[tr_idx], y_v[tr_idx])
        # 訓練 Arousal 模型
        model_a.fit(text[tr_idx], y_a[tr_idx])
        # 對驗證集進行預測
        preds_v[va_idx] = model_v.predict(text[va_idx])
        preds_a[va_idx] = model_a.predict(text[va_idx])

    # === 印出交叉驗證的結果 ===
    print("[Train-CV] Valence  Pearson:", round(pearson(y_v, preds_v), 4),
          " MSE:", round(mean_squared_error(y_v, preds_v), 4))
    print("[Train-CV] Arousal  Pearson:", round(pearson(y_a, preds_a), 4),
          " MSE:", round(mean_squared_error(y_a, preds_a), 4))

    # === 使用全部訓練資料重新訓練最終模型 ===
    model_v.fit(text, y_v)
    model_a.fit(text, y_a)

    # === 若有開發集 (dev set) 則進行預測與評估 ===
    if dev_path:
        df_dev = pd.read_csv(dev_path)
        x_dev = df_dev["text"].astype(str).values
        pv = model_v.predict(x_dev)  # 預測 Valence
        pa = model_a.predict(x_dev)  # 預測 Arousal

        # 若開發集有標籤，計算 Pearson 與 MSE
        if "valence" in df_dev and "arousal" in df_dev:
            print("[Dev] Valence   Pearson:", round(pearson(df_dev["valence"].values, pv), 4),
                  " MSE:", round(mean_squared_error(df_dev["valence"].values, pv), 4))
            print("[Dev] Arousal   Pearson:", round(pearson(df_dev["arousal"].values, pa), 4),
                  " MSE:", round(mean_squared_error(df_dev["arousal"].values, pa), 4))

        # 若指定要輸出預測結果，則輸出成 CSV 檔案
        if out_pred_path:
            out = df_dev[["id"]].copy()
            out["valence"] = pv
            out["arousal"] = pa
            out.to_csv(out_pred_path, index=False)
            print(f"[Saved] {out_pred_path}")

# === 主程式入口 ===
if __name__ == "__main__":
    ap = argparse.ArgumentParser()
    ap.add_argument("--train", required=True, help="path to train.csv")  # 訓練資料路徑
    ap.add_argument("--dev", required=False, help="path to dev.csv (labels optional)")  # 驗證/測試資料
    ap.add_argument("--out", required=False, help="path to save predictions on dev/test")  # 輸出預測檔案
    args = ap.parse_args()
    train_eval(args.train, args.dev, args.out)
